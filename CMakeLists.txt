cmake_minimum_required(VERSION 3.24)
project(audio_obs_asio VERSION 0.1.0 LANGUAGES CXX)

add_library(audio_mixer src/mixer.cpp)
target_include_directories(audio_mixer PUBLIC include)
target_compile_features(audio_mixer PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(audio_mixer PRIVATE /W4 /WX)
else()
  target_compile_options(audio_mixer PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(mixer_tests tests/mixer_tests.cpp)
  target_link_libraries(mixer_tests PRIVATE audio_mixer)
  add_test(NAME mixer_routing COMMAND mixer_tests)
  find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
  add_test(NAME osc_transport COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -p test_osc.py -v)
  set_tests_properties(osc_transport PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
