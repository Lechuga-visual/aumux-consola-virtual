cmake_minimum_required(VERSION 3.24)
project(aumux_audio VERSION 0.12.0 LANGUAGES CXX)

option(AUMUX_EDITOR "Build the AuMux-styled source editor" ON)
find_path(OBS_INCLUDE_DIR obs-module.h PATH_SUFFIXES obs REQUIRED)
find_library(OBS_LIBRARY NAMES obs libobs REQUIRED)
add_library(monitor-mix-receiver MODULE ../obs-receiver.cpp)
target_compile_features(monitor-mix-receiver PRIVATE cxx_std_17)
target_include_directories(monitor-mix-receiver PRIVATE "${OBS_INCLUDE_DIR}")
target_link_libraries(monitor-mix-receiver PRIVATE "${OBS_LIBRARY}")
set_target_properties(monitor-mix-receiver PROPERTIES PREFIX "")
if(AUMUX_EDITOR)
  target_sources(monitor-mix-receiver PRIVATE ../obs-receiver-ui.cpp)
  target_compile_definitions(monitor-mix-receiver PRIVATE AUMUX_WITH_UI)
  if(WIN32)
    target_link_libraries(monitor-mix-receiver PRIVATE user32 gdi32 comctl32)
  else()
    find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets)
    target_link_libraries(monitor-mix-receiver PRIVATE Qt6::Widgets)
  endif()
endif()
if(WIN32)
  target_link_libraries(monitor-mix-receiver PRIVATE ws2_32)
  install(TARGETS monitor-mix-receiver RUNTIME DESTINATION obs-plugins/64bit LIBRARY DESTINATION obs-plugins/64bit)
elseif(APPLE)
  # OBS expects a .plugin bundle, with its binary in Contents/MacOS.
  set_target_properties(monitor-mix-receiver PROPERTIES BUNDLE TRUE BUNDLE_EXTENSION plugin)
  install(TARGETS monitor-mix-receiver BUNDLE DESTINATION . LIBRARY DESTINATION .)
else()
  install(TARGETS monitor-mix-receiver LIBRARY DESTINATION lib/obs-plugins)
endif()

option(AUMUX_TEST_EDITOR "Build the real libobs/Qt editor smoke test" OFF)
if(AUMUX_TEST_EDITOR AND AUMUX_EDITOR AND NOT WIN32)
  add_executable(aumux-editor-test ../obs-receiver.cpp ../obs-receiver-ui.cpp ../obs-ui-test.cpp)
  target_compile_features(aumux-editor-test PRIVATE cxx_std_17)
  target_compile_definitions(aumux-editor-test PRIVATE AUMUX_WITH_UI)
  target_include_directories(aumux-editor-test PRIVATE "${OBS_INCLUDE_DIR}")
  target_link_libraries(aumux-editor-test PRIVATE "${OBS_LIBRARY}" Qt6::Widgets)
endif()
